# 苏州园林数字资产安全分发与溯源系统

## 技术架构设计文档 v1.0

---

## 一、项目概述

### 1.1 项目背景

苏州园林作为世界文化遗产，其数字化资产（高清影像、VR模型、设计图纸、文献资料）面临：
- 版权归属不清，原创者权益难以保障
- 资产分发后无法追踪去向
- 侵权行为难以发现和取证
- 授权流程缺乏可信记录

### 1.2 解决方案

构建"水印隐形身份证 + 区块链不可篡改出生证 + AI智能巡查"三位一体的数字资产保护体系，实现：
- **确权**：上传即存证，版权归属链上可查
- **用权**：授权流程数字化，智能合约自动执行
- **维权**：AI巡查侵权，区块链提供司法级证据

### 1.3 核心创新

| 创新点 | 说明 |
|-------|------|
| 自研鲁棒水印算法 | 抗几何攻击的矢量/栅格水印嵌入与提取 |
| 水印+区块链双保险 | 水印携带资产ID，链上记录完整生命周期 |
| AI智能侵权检测 | 多平台爬虫 + 图像相似度比对 + GLM智能分析 |
| 全链路溯源 | 知识图谱构建资产流转关系网络 |

#### 3.2.3 水印ID编码规则

```
水印ID格式：GARDEN-ASSET-TIMESTAMP-CHECKSUM

示例：ZY-IMG-20260315143522-A3F8

解析：
- ZY: 拙政园代码（园林标识）
- IMG: 影像类型（资产类型）
- 20260315143522: 嵌入时间戳
- A3F8: 校验码（4位十六进制）

水印ID用途：
1. 嵌入到资产中（隐形身份证）
2. 写入区块链（链上存证）
3. 侵权检测时提取并溯源
```

#### 3.3.2 智能合约设计

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

/**
 * @title 资产存证合约
 * @dev 记录数字资产的上链存证信息
 */
contract AssetRegistry {

    // 存证记录结构
    struct AssetRecord {
        string assetCode;           // 资产编码
        string watermarkId;         // 水印ID
        string fileHash;            // 文件哈希(SHA256)
        string featureHash;         // 特征向量哈希
        address creator;            // 创建者地址
        uint256 timestamp;          // 存证时间戳
        uint8 status;               // 状态：0-已存证 1-已授权 2-已归档
    }

    // 授权记录结构
    struct LicenseRecord {
        string assetCode;           // 资产编码
        address licensor;           // 授权方
        address licensee;           // 被授权方
        uint8 licenseType;          // 授权类型
        uint256 startTime;          // 授权开始时间
        uint256 endTime;            // 授权结束时间
        string usageScope;          // 使用范围
        uint256 timestamp;          // 记录时间
    }

    // 存证记录映射
    mapping(string => AssetRecord) public assetRecords;
    mapping(string => LicenseRecord[]) public licenseHistory;

    // 事件
    event AssetRegistered(string indexed assetCode, string watermarkId, address creator);
    event LicenseGranted(string indexed assetCode, address licensee, uint8 licenseType);

    /**
     * @dev 资产存证
     */
    function registerAsset(
        string memory _assetCode,
        string memory _watermarkId,
        string memory _fileHash,
        string memory _featureHash
    ) public returns (bool) {
        require(bytes(assetRecords[_assetCode].assetCode).length == 0, "Asset already registered");

        assetRecords[_assetCode] = AssetRecord({
            assetCode: _assetCode,
            watermarkId: _watermarkId,
            fileHash: _fileHash,
            featureHash: _featureHash,
            creator: msg.sender,
            timestamp: block.timestamp,
            status: 0
        });

        emit AssetRegistered(_assetCode, _watermarkId, msg.sender);
        return true;
    }

    /**
     * @dev 授权记录
     */
    function grantLicense(
        string memory _assetCode,
        address _licensee,
        uint8 _licenseType,
        uint256 _startTime,
        uint256 _endTime,
        string memory _usageScope
    ) public returns (bool) {
        require(bytes(assetRecords[_assetCode].assetCode).length > 0, "Asset not registered");
        require(assetRecords[_assetCode].creator == msg.sender, "Not the creator");

        licenseHistory[_assetCode].push(LicenseRecord({
            assetCode: _assetCode,
            licensor: msg.sender,
            licensee: _licensee,
            licenseType: _licenseType,
            startTime: _startTime,
            endTime: _endTime,
            usageScope: _usageScope,
            timestamp: block.timestamp
        }));

        emit LicenseGranted(_assetCode, _licensee, _licenseType);
        return true;
    }

    /**
     * @dev 查询资产存证
     */
    function getAssetRecord(string memory _assetCode) public view returns (AssetRecord memory) {
        return assetRecords[_assetCode];
    }

    /**
     * @dev 验证水印ID
     */
    function verifyWatermark(string memory _assetCode, string memory _watermarkId) public view returns (bool) {
        return keccak256(bytes(assetRecords[_assetCode].watermarkId)) == keccak256(bytes(_watermarkId));
    }
}
```

#### 3.4.4 向量检索服务

```python
# ai_service/vector_store.py

from pymilvus import MilvusClient, DataType
import numpy as np
from typing import List, Optional
from PIL import Image
import torch
from transformers import CLIPProcessor, CLIPModel

class VectorSearchService:
    """向量检索服务"""

    def __init__(self, collection_name: str = "garden_assets"):
        self.client = MilvusClient("milvus_demo.db")  # Lite版本
        self.collection_name = collection_name
        self.clip_model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        self.clip_processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")

        self._init_collection()

    def _init_collection(self):
        """初始化向量集合"""
        if not self.client.has_collection(self.collection_name):
            self.client.create_collection(
                collection_name=self.collection_name,
                dimension=512  # CLIP向量维度
            )

    def extract_features(self, image: Image.Image) -> np.ndarray:
        """提取图像特征向量"""
        inputs = self.clip_processor(images=image, return_tensors="pt")

        with torch.no_grad():
            features = self.clip_model.get_image_features(**inputs)

        # 归一化
        features = features / features.norm(dim=-1, keepdim=True)
        return features.numpy().flatten()

    def insert_asset(self, asset_id: str, image: Image.Image, metadata: dict):
        """插入资产向量"""
        vector = self.extract_features(image)

        self.client.insert(
            collection_name=self.collection_name,
            data=[{
                "id": asset_id,
                "vector": vector.tolist(),
                "metadata": metadata
            }]
        )

    def search_similar(self, query_image: Image.Image, top_k: int = 10) -> List[dict]:
        """搜索相似图像"""
        query_vector = self.extract_features(query_image)

        results = self.client.search(
            collection_name=self.collection_name,
            data=[query_vector.tolist()],
            limit=top_k,
            output_fields=["metadata"]
        )

        return results[0]

    def check_infringement(self, query_image: Image.Image, threshold: float = 0.85) -> dict:
        """侵权检测"""
        similar_results = self.search_similar(query_image, top_k=5)

        # 筛选超过阈值的
        infringements = []
        for result in similar_results:
            if result['distance'] >= threshold:
                infringements.append({
                    'asset_id': result['id'],
                    'similarity': result['distance'],
                    'metadata': result['entity']['metadata']
                })

        return {
            'has_infringement': len(infringements) > 0,
            'matches': infringements,
            'risk_level': 'high' if infringements else 'low'
        }
```
